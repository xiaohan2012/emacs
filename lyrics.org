* lyrics panel for simple-mpc

Lyrics of the playing song in a right-half side window; =L= in simple-mpc toggles it.

*Keys (panel):*
- =g= reload
- =e= edit cached file (wrong lyrics, misaligned timestamps); panel reloads on save
- =s= pick lyrics: edit the search, choose a version (title - singers - length - source)
- =R= drop cache, fetch again
- =+= / =-= show lyrics 0.5s later / earlier; saved as =[offset:]= in the =.lrc=
- =q= close (also stops the mpc watcher)
- =?= list keys
- hints also shown at panel top; =L= listed in simple-mpc's main screen

*Font:* =my/lyrics-font-family= (LXGW WenKai, a Kaiti close to handwriting; =brew install --cask font-lxgw-wenkai=) at =my/lyrics-font-scale= (1.5×), panel only.

*Lyrics source:*
- local =.lrc= / =.txt= next to the song first
- else online, then cached next to the song (offline afterwards)
  - synced → =.lrc=, plain → =.txt=
- *NetEase* first, *QQ Music* second: both strong on Chinese songs, each backs up the other
- QQ also has Jay Chou's catalogue and some Finnish songs NetEase lacks, but its API often answers error 2001
- cached files are tagged =[ti:]= =[ar:]= =[re:]= with the version used; the panel shows it under the title
  - warning face when the singers don't include the song's artist → likely a different song, press =s=

*Synced scrolling (=.lrc= only):*
- current line highlighted and kept centered
- position read from MPD's =status= (ms precision) on each player event, extrapolated between events
- you scroll or move point in the panel → auto-scroll pauses 5s

*Matching:*
- no tags on org-songs files → artist = parent dir, title = file name
- folder may not be an artist (e.g. =未归档=) → search title + artist, then title alone
- hit must match the title; prefer same artist > exact title > non-live

  (defun my/lyrics--playing-song ()
    "Return (FILE ARTIST TITLE) of the playing song, or nil if nothing plays."
    (let ((out (string-trim-right
                (shell-command-to-string
                 "mpc -f '%file%\\t%artist%\\t%title%' current"))))
      (unless (string-empty-p out)
        (pcase-let ((`(,file ,artist ,title) (split-string out "\t")))
          (list file
                (if (string-empty-p (or artist ""))
                    (file-name-nondirectory (directory-file-name (or (file-name-directory file) "")))
                  artist)
                (if (string-empty-p (or title ""))
                    (file-name-base file)
                  title))))))

  (defun my/lyrics--local-file (file)

  (defun my/lyrics--strip-timestamps (text)
    "Remove LRC timestamps and ID tag lines such as [ti:...] from TEXT."
    (replace-regexp-in-string
     "^\\(\\[[0-9:.]+\\]\\)+" ""
     (replace-regexp-in-string "^\\[[a-z]+:[^]\n]*\\]\n" "" text)))

  (defun my/lyrics--read-local (path)
    (with-temp-buffer
      (insert-file-contents path)
      (buffer-string)))

  (defun my/lyrics--parse-lrc (text)
    "Parse LRC TEXT into (OFFSET . ENTRIES).
  OFFSET is the [offset:] tag in seconds; ENTRIES is a time-sorted list of (TIME . LINE), nil when TEXT has no timestamps.
  A line with several timestamps yields one entry per timestamp; an untimed line inherits the previous time."
    (let ((offset 0) (last 0) (timed nil) (entries nil))
      (dolist (line (split-string text "\n"))
        (cond
         ((string-match "\\`\\[offset:[ \t]*\\([-+]?[0-9]+\\)\\]" line)
          (setq offset (/ (string-to-number (match-string 1 line)) 1000.0)))
         ((string-match-p "\\`\\[[a-z]+:" line))
         (t
          (let ((pos 0) (times nil))
            (while (and (string-match "\\[\\([0-9]+\\):\\([0-9.]+\\)\\]" line pos)
                        (= (match-beginning 0) pos))
              (push (+ (* 60 (string-to-number (match-string 1 line)))
                       (string-to-number (match-string 2 line)))
                    times)
              (setq pos (match-end 0)))
            (let ((text (substring line pos)))
              (if times
                  (progn
                    (dolist (time times) (push (cons time text) entries))
                    (setq timed t last (apply #'max times)))
                (unless (string-blank-p text)
                  (push (cons last text) entries))))))))
      (cons offset (and timed (seq-sort-by #'car #'< (nreverse entries))))))

  (defun my/lyrics--source-line (text)
    "Describe the lyrics version in TEXT from the tags `my/lyrics--save-cache' writes, or nil if untagged."
    (let ((tag (lambda (name)
                 (when (string-match (format "^\\[%s:\\([^]\n]*\\)\\]" name) text)
                   (match-string 1 text)))))
      (let ((source (funcall tag "re")))
        (when (member source '("netease" "qq"))
          (let* ((singers (or (funcall tag "ar") ""))
                 (mismatch (not (member (nth 1 my/lyrics--current-song) (split-string singers "/")))))
            (propertize (format "来源：%s · %s - %s%s"
                                (if (equal source "qq") "QQ音乐" "网易云")
                                singers (or (funcall tag "ti") "")
                                (if mismatch "（歌手不一致，按 s 换一个）" ""))
                        'face (if mismatch 'warning 'shadow)))))))

  (defun my/lyrics--render (header lyrics)
    "Show HEADER and LYRICS, raw LRC or plain text; synced LRC sets up `my/lyrics--lines'."
    (with-current-buffer (get-buffer-create my/lyrics-buffer)
      (unless (derived-mode-p 'my/lyrics-mode) (my/lyrics-mode))
      (let ((inhibit-read-only t)
            (parsed (my/lyrics--parse-lrc (or lyrics "")))
            (lines nil))
        (erase-buffer)
        (when (overlayp my/lyrics--overlay) (delete-overlay my/lyrics--overlay))
        (insert (propertize header 'face 'bold) "\n"
                (let ((source (my/lyrics--source-line (or lyrics ""))))
                  (if source (concat source "\n") ""))
                (propertize "g 刷新  e 编辑  s 选歌词  R 重新获取  +/- 推迟/提前  q 关闭  ? 帮助" 'face 'shadow) "\n\n")
        (cond ((cdr parsed)
               (dolist (entry (cdr parsed))
                 (push (cons (car entry) (point)) lines)
                 (insert (cdr entry) "\n")))
              (lyrics (insert (my/lyrics--strip-timestamps lyrics)))
              (t (insert "(未找到歌词，按 e 手动添加)")))
        (setq my/lyrics--lines (and lines (vconcat (nreverse lines)))
              my/lyrics--offset (car parsed)
              my/lyrics--index nil
              my/lyrics--auto-view nil
              my/lyrics--need-scroll t)
        (goto-char (point-min)))))

  (defun my/lyrics--get-json (url callback &optional body)
    "Fetch URL and call CALLBACK with its parsed JSON, or nil on any failure.
  With BODY, a JSON-serializable object, send it as a POST request."
    (let ((url-request-method (if body "POST" "GET"))
          (url-request-data (when body (encode-coding-string (json-serialize body) 'utf-8)))
          ;; Set the agent here, not as an extra header: url.el would then send two User-Agent headers, which QQ Music rejects.
          (url-user-agent "Mozilla/5.0")
          (url-request-extra-headers `(;; QQ Music rejects requests without its own Referer.
                                       ("Referer" . "https://y.qq.com")
                                       ,@(when body '(("Content-Type" . "application/json"))))))
      (url-retrieve
       url
       (lambda (status)
         (let ((json (ignore-errors
                       (unless (plist-get status :error)
                         (goto-char url-http-end-of-headers)
                         (json-parse-buffer :object-type 'alist :array-type 'list
                                            :null-object nil :false-object nil)))))
           (kill-buffer)
           (funcall callback json)))
       ;; Inhibit cookies: once NetEase's NMTID cookie is sent back, its search returns unrelated songs.
       nil t t)))

  (defun my/lyrics--first (fetchers callback)
    "Run async FETCHERS in order until one yields non-nil, then call CALLBACK with it.
  Each fetcher takes a continuation; CALLBACK gets nil when all fail."
    (if (null fetchers)
        (funcall callback nil)
      (funcall (car fetchers)
               (lambda (result)
                 (if result
                     (funcall callback result)
                   (my/lyrics--first (cdr fetchers) callback))))))

  (defun my/lyrics--normalize (name)
    "Lowercase NAME without parenthesized parts and punctuation, for fuzzy title matching."
    (downcase (replace-regexp-in-string
               "[^[:alnum:]]+" ""
               (replace-regexp-in-string "[(（][^)）]*[)）]" "" name))))

  (defun my/lyrics--score (name singers artist title)
    "Rank a search hit called NAME by SINGERS for ARTIST and TITLE; nil when NAME does not match TITLE."
    (let ((norm (my/lyrics--normalize name))
          (want (my/lyrics--normalize title)))
      (when (and (> (length norm) 0) (> (length want) 0)
                 (or (string-search want norm) (string-search norm want)))
        (+ (if (member artist singers) 4 0)
           (if (equal norm want) 2 0)
           (if (string-match-p "live" (downcase name)) 0 1)))))

  (defun my/lyrics--netease-search (query callback)
    "Search NetEase Cloud Music for QUERY; call CALLBACK with a list of hits.
  A hit is a plist (:source :id :title :singers :duration), :duration in seconds."
    (my/lyrics--get-json
     (concat "https://music.163.com/api/search/get?"
             (url-build-query-string `((s ,query) (type 1) (limit 10))))
     (lambda (json)
       (funcall callback
                (mapcar (lambda (song)
                          (list :source 'netease
                                :id (alist-get 'id song)
                                :title (alist-get 'name song)
                                :singers (mapcar (lambda (a) (alist-get 'name a)) (alist-get 'artists song))
                                :duration (/ (or (alist-get 'duration song) 0) 1000)))
                        (alist-get 'songs (alist-get 'result json)))))))

  (defun my/lyrics--qq-search (query callback)
    "Search QQ Music for QUERY; call CALLBACK with a list of hits, see `my/lyrics--netease-search'."
    (my/lyrics--get-json
     "https://u.y.qq.com/cgi-bin/musicu.fcg"
     (lambda (json)
       (funcall callback
                (mapcar (lambda (song)
                          (list :source 'qq
                                :id (alist-get 'mid song)
                                :title (alist-get 'name song)
                                :singers (mapcar (lambda (a) (alist-get 'name a)) (alist-get 'singer song))
                                :duration (or (alist-get 'interval song) 0)))
                        (let-alist json .req.data.body.song.list))))
     `((req . ((module . "music.search.SearchCgiService")
               (method . "DoSearchForQQMusicDesktop")
               (param . ((query . ,query) (num_per_page . 10) (page_num . 1) (search_type . 0))))))))

  (defun my/lyrics--fetch-lyric (hit callback)
    "Call CALLBACK with HIT's lyrics as a result plist (HIT plus :lrc), or nil when it has none."
    (let ((done (lambda (lrc)
                  (funcall callback (when (and lrc (not (string-blank-p lrc)))
                                      (append (list :lrc lrc) hit))))))
      (pcase (plist-get hit :source)
        ('netease
         (my/lyrics--get-json
          (format "https://music.163.com/api/song/lyric?id=%s&lv=1" (plist-get hit :id))
          (lambda (json) (funcall done (alist-get 'lyric (alist-get 'lrc json))))))
        ('qq
         (my/lyrics--get-json
          (concat "https://c.y.qq.com/lyric/fcgi-bin/fcg_query_lyric_new.fcg?"
                  (url-build-query-string `((songmid ,(plist-get hit :id)) (format "json") (nobase64 1))))
          (lambda (json) (funcall done (alist-get 'lyric json))))))))

  (defun my/lyrics--auto (search query artist title)
    "Return a fetcher running SEARCH for QUERY and trying hits that match TITLE, best first."
    (lambda (callback)
      (funcall search query
               (lambda (hits)
                 (let ((scored (delq nil (mapcar (lambda (hit)
                                                   (let ((score (my/lyrics--score (plist-get hit :title)
                                                                                  (plist-get hit :singers)
                                                                                  artist title)))
                                                     (when score (cons score hit))))
                                                 hits))))
                   (my/lyrics--first
                    (mapcar (lambda (scored-hit) (lambda (k) (my/lyrics--fetch-lyric (cdr scored-hit) k)))
                            (seq-sort-by #'car #'> scored))
                    callback))))))

  (defun my/lyrics--fetch-online (artist title callback)
    "Call CALLBACK with a result plist for ARTIST and TITLE, or nil if no source has lyrics."
    (my/lyrics--first
     (list (my/lyrics--auto #'my/lyrics--netease-search (format "%s %s" title artist) artist title)
           (my/lyrics--auto #'my/lyrics--netease-search title artist title)
           (my/lyrics--auto #'my/lyrics--qq-search (format "%s %s" title artist) artist title)
           (my/lyrics--auto #'my/lyrics--qq-search title artist title))
     callback))

  (defun my/lyrics--save-cache (file result)
    "Write RESULT's lyrics next to song FILE, tagged with where they came from; return the path.
  Synced lyrics go to .lrc, plain ones to .txt; any other cached file of the song is removed."
    (let* ((lrc (plist-get result :lrc))
           (synced (string-match-p "^\\[[0-9]" lrc))
           (base (my/lyrics--song-base file))
           (path (concat base (if synced ".lrc" ".txt"))))
      (dolist (ext '(".lrc" ".txt"))
        (when (file-exists-p (concat base ext)) (delete-file (concat base ext))))
      (with-temp-file path
        (insert (format "[ti:%s]\n[ar:%s]\n[re:%s]\n"
                        (plist-get result :title)
                        (string-join (plist-get result :singers) "/")
                        (plist-get result :source))
                lrc))
      path))

  (defun my/lyrics--show (song &optional skip-local)
    "Display lyrics of SONG, a (FILE ARTIST TITLE) list; with SKIP-LOCAL ignore the cache."
    (pcase-let* ((`(,file ,artist ,title) song)
                 (header (format "%s - %s" artist title))
                 (local (unless skip-local (my/lyrics--local-file file))))
      (setq my/lyrics--current-song song)
      (if local
          (my/lyrics--render header (my/lyrics--read-local local))
        (my/lyrics--render header "加载中...")
        (my/lyrics--fetch-online
         artist title
         (lambda (result)
           ;; Drop stale responses if the song changed meanwhile.
           (when (equal song my/lyrics--current-song)
             (if result
                 (my/lyrics--render header (my/lyrics--read-local (my/lyrics--save-cache file result)))
               (my/lyrics--render header nil))))))))

  (defun my/lyrics--wait (fn)
    "Call async FN with a continuation and block until it delivers, at most 10s; return the value."
    (let ((done nil) (value nil) (deadline (+ (float-time) 10)))
      (funcall fn (lambda (v) (setq value v done t)))
      (while (and (not done) (< (float-time) deadline))
        (accept-process-output nil 0.05))
      value))

  (defun my/lyrics--mmss (seconds)
    (format "%d:%02d" (/ seconds 60) (% seconds 60)))

  (defun my/lyrics-pick ()
    "Search lyrics with an editable query and choose which version to cache for the shown song."
    (interactive)
    (pcase-let ((`(,file ,artist ,title) my/lyrics--current-song))
      (unless file (user-error "No song is playing"))
      (let* ((query (read-string "搜索歌词: " (format "%s %s" title artist)))
             (hits (append (my/lyrics--wait (lambda (k) (my/lyrics--netease-search query k)))
                           (my/lyrics--wait (lambda (k) (my/lyrics--qq-search query k)))))
             (duration (round (string-to-number (or (cdr (assoc "duration" (my/lyrics--mpd-status))) "0"))))
             (choices (seq-map-indexed
                       (lambda (hit i)
                         (cons (format "%2d. %s - %s  %s  [%s]" (1+ i)
                                       (plist-get hit :title)
                                       (string-join (plist-get hit :singers) "/")
                                       (my/lyrics--mmss (plist-get hit :duration))
                                       (if (eq (plist-get hit :source) 'qq) "QQ音乐" "网易云"))
                               hit))
                       hits)))
        (unless hits (user-error "没有搜到，换个关键词试试"))
        (let* ((label (completing-read
                       (format "选择歌词（这首歌 %s）: " (my/lyrics--mmss duration))
                       ;; Keep the search order instead of letting the completion UI sort.
                       (lambda (string pred action)
                         (if (eq action 'metadata)
                             '(metadata (display-sort-function . identity))
                           (complete-with-action action choices string pred)))
                       nil t))
               (result (my/lyrics--wait (lambda (k) (my/lyrics--fetch-lyric (cdr (assoc label choices)) k)))))
          (unless result (user-error "这个版本没有歌词，换一个"))
          (my/lyrics--save-cache file result)
          (my/lyrics--show my/lyrics--current-song)))))

  (defun my/lyrics-refresh (&optional force)
    "Show lyrics of the playing song; skip when the song is unchanged unless FORCE."
    (interactive (list t))
    (let ((song (my/lyrics--playing-song)))
      (cond ((null song)
             (setq my/lyrics--current-song nil)
             (my/lyrics--render "未在播放" ""))
            ((or force (not (equal song my/lyrics--current-song)))
             (my/lyrics--show song)))))

  (defun my/lyrics-edit ()
    "Edit the lyrics file of the shown song, creating a .lrc if none exists."
    (interactive)
    (let ((file (car my/lyrics--current-song)))
      (unless file (user-error "No song is playing"))
      (find-file-other-window
       (or (my/lyrics--local-file file) (concat (my/lyrics--song-base file) ".lrc")))
      (add-hook 'after-save-hook (lambda () (my/lyrics-refresh t)) nil t)))

  (defun my/lyrics-refetch ()
    "Delete the cached lyrics of the shown song and fetch them again online."
    (interactive)
    (let* ((song my/lyrics--current-song)
           (local (and song (my/lyrics--local-file (car song)))))
      (unless song (user-error "No song is playing"))
      (when (and local (not (yes-or-no-p (format "Overwrite %s? " (file-name-nondirectory local)))))
        (user-error "Cancelled"))
      (when local (delete-file local))
      (my/lyrics--show song t)))

  (defun my/lyrics--mpd-status ()
    "Return MPD's status as an alist of (FIELD . VALUE) strings, or nil when MPD is unreachable.
  Talks to MPD directly because `mpc status' rounds the elapsed time to whole seconds."
    (ignore-errors
      (with-temp-buffer
        (let ((proc (open-network-stream "mpd-status" (current-buffer)
                                         (or (getenv "MPD_HOST") "localhost")
                                         (string-to-number (or (getenv "MPD_PORT") "6600"))))
              (deadline (+ (float-time) 1))
              (fields nil))
          (set-process-query-on-exit-flag proc nil)
          (process-send-string proc "status\nclose\n")
          (while (and (process-live-p proc) (< (float-time) deadline))
            (accept-process-output proc 0.05))
          (delete-process proc)
          (goto-char (point-min))
          (while (re-search-forward "^\\([a-z]+\\): \\(.*\\)$" nil t)
            (push (cons (match-string 1) (match-string 2)) fields))
          fields))))

  (defun my/lyrics--sync-clock ()
    "Re-anchor the playback clock to MPD's elapsed time and state."
    (let* ((status (my/lyrics--mpd-status))
           (elapsed (string-to-number (or (cdr (assoc "elapsed" status)) "0"))))
      (setq my/lyrics--clock (if (equal (cdr (assoc "state" status)) "play")
                                 (cons 'play (- (float-time) elapsed))
                               (cons 'pause elapsed))
            my/lyrics--clock-synced-at (float-time))
      (my/lyrics--tick)))

  (defun my/lyrics--elapsed ()
    (pcase my/lyrics--clock
      (`(play . ,anchor) (- (float-time) anchor))
      (`(pause . ,elapsed) elapsed)
      (_ 0)))

  (defun my/lyrics--line-index (time)
    "Index of the last line in `my/lyrics--lines' starting at or before TIME, or nil."
    (let ((index nil))
      (seq-do-indexed (lambda (line i) (when (<= (car line) time) (setq index i)))
                      my/lyrics--lines)
      index))

  (defun my/lyrics--tick ()
    "Highlight the line being sung and keep it centered unless the user recently scrolled."
    (let ((win (get-buffer-window my/lyrics-buffer)))
      (when (and win my/lyrics--lines
                 (> (- (float-time) my/lyrics--clock-synced-at) my/lyrics-resync-interval))
        ;; Calls back into this function with a fresh clock.
        (my/lyrics--sync-clock))
      (when (and win my/lyrics--lines)
        (with-current-buffer my/lyrics-buffer
          (let ((now (float-time))
                (index (my/lyrics--line-index (+ (my/lyrics--elapsed) my/lyrics--offset)))
                (view (cons (window-start win) (window-point win))))
            (when (and my/lyrics--auto-view (not (equal view my/lyrics--auto-view)))
              (setq my/lyrics--follow-paused-until (+ now my/lyrics-follow-pause)
                    my/lyrics--auto-view view
                    my/lyrics--need-scroll t))
            (unless (eql index my/lyrics--index)
              (setq my/lyrics--index index
                    my/lyrics--need-scroll t)
              (unless (overlayp my/lyrics--overlay)
                (setq my/lyrics--overlay (make-overlay 1 1))
                (overlay-put my/lyrics--overlay 'face 'my/lyrics-current-line))
              (if index
                  (let ((start (cdr (aref my/lyrics--lines index))))
                    (move-overlay my/lyrics--overlay start
                                  (save-excursion (goto-char start) (line-end-position))
                                  (current-buffer)))
                (delete-overlay my/lyrics--overlay)))
            (when (and my/lyrics--need-scroll (>= now my/lyrics--follow-paused-until))
              (let ((target (cdr (aref my/lyrics--lines (or index 0)))))
                (set-window-point win target)
                ;; Set the start explicitly rather than `recenter', which only applies at redisplay
                ;; and would make the recorded view differ from the real one.
                (set-window-start win (save-excursion
                                        (goto-char target)
                                        ;; Count lines in the panel's scaled font, not the frame's default.
                                        (forward-line (- (/ (window-body-height win t)
                                                            (with-selected-window win (default-line-height))
                                                            2)))
                                        (point)))
                (setq my/lyrics--auto-view (cons (window-start win) (window-point win))
                      my/lyrics--need-scroll nil))))))))

  (defun my/lyrics--shift (delta)
    "Show lyrics DELTA seconds earlier and save the total as the .lrc [offset:] tag."
    (let ((path (and my/lyrics--current-song (my/lyrics--local-file (car my/lyrics--current-song)))))
      (unless (and my/lyrics--lines path (string-suffix-p ".lrc" path))
        (user-error "No synced lyrics to shift"))
      (setq my/lyrics--offset (+ my/lyrics--offset delta))
      (let ((ms (round (* 1000 my/lyrics--offset))))
        (with-temp-file path
          (insert-file-contents path)
          (goto-char (point-min))
          (if (re-search-forward "^\\[offset:[^]\n]*\\]" nil t)
              (replace-match (format "[offset:%d]" ms) t t)
            (insert (format "[offset:%d]\n" ms))))
        (my/lyrics--tick)
        (message (if (= ms 0) "歌词时间已复原"
                   (format "歌词%s %.1fs" (if (> ms 0) "提前" "推迟") (abs (/ ms 1000.0))))))))

  (defun my/lyrics-delay ()
    "Show synced lyrics 0.5s later."
    (interactive)
    (my/lyrics--shift -0.5))

  (defun my/lyrics-advance ()
    "Show synced lyrics 0.5s earlier."
    (interactive)
    (my/lyrics--shift 0.5))

  (defun my/lyrics--start-watch ()
    "Refresh lyrics and the playback clock whenever MPD's player state changes."
    (unless (process-live-p my/lyrics--idle-proc)
      (setq my/lyrics--idle-proc
            (make-process :name "mpc-idle"
                          :command '("mpc" "idleloop" "player")
                          :noquery t
                          :filter (lambda (_proc _out)
                                    (my/lyrics-refresh)
                                    (my/lyrics--sync-clock)))))
    (unless (timerp my/lyrics--timer)
      (setq my/lyrics--timer (run-at-time 0 0.2 #'my/lyrics--tick))))

  (defun my/lyrics--stop-watch ()
    (when (process-live-p my/lyrics--idle-proc)
      (delete-process my/lyrics--idle-proc))
    (when (timerp my/lyrics--timer)
      (cancel-timer my/lyrics--timer)
      (setq my/lyrics--timer nil)))

  (defun my/lyrics-toggle ()
    "Toggle the lyrics panel on the right half of the frame."
    (interactive)
    (let ((win (get-buffer-window my/lyrics-buffer)))
      (if win
          (progn (delete-window win) (my/lyrics--stop-watch))
        (my/lyrics-refresh t)
        (display-buffer (get-buffer-create my/lyrics-buffer)
                        '(display-buffer-in-side-window (side . right) (window-width . 0.5)))
        (my/lyrics--sync-clock)
        (my/lyrics--start-watch))))

  (with-eval-after-load 'simple-mpc-mode
    (define-key simple-mpc-mode-map (kbd "L") #'my/lyrics-toggle))

  (defun my/lyrics--add-to-main-screen (&rest _)
    "List the lyrics key in simple-mpc's hard-coded main screen menu."
    (when-let ((buf (get-buffer simple-mpc-main-buffer-name)))
      (with-current-buffer buf
        (let ((inhibit-read-only t))
          (goto-char (point-max))
          (insert (propertize "\n\n   * lyrics\n" 'face 'simple-mpc-main-headers)
                  "      * toggle [L]yrics panel")
          (goto-char (point-min))))))

  (advice-add 'simple-mpc :after #'my/lyrics--add-to-main-screen)

  ;; Re-loading this file applies font changes to an open panel.
  (when (get-buffer my/lyrics-buffer)
    (with-current-buffer my/lyrics-buffer (my/lyrics--apply-font)))

  ;; Re-loading this file must not leave a watcher or timer running old closures.
  (when (or (process-live-p my/lyrics--idle-proc) (timerp my/lyrics--timer))
    (my/lyrics--stop-watch)
    (my/lyrics--start-watch))
