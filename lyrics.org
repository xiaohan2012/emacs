* lyrics panel for simple-mpc

Show lyrics of the playing song in a side window on the right half of the frame.
Press =L= in simple-mpc to toggle the panel.

Lyrics are read from a =.lrc= or =.txt= file next to the song.
When there is none, they are fetched online and cached next to the song, so later plays work offline.
NetEase Cloud Music is tried first, QQ Music second: both cover Chinese songs well and back each other up when one API breaks.
QQ Music also has Jay Chou's catalogue and some Finnish songs NetEase lacks.
Synced lyrics are cached as =.lrc= (timestamps kept for a future synced view), plain lyrics as =.txt=.

Keys in the lyrics panel:
- =g= reload the current song's lyrics
- =e= edit the cached file, e.g. to fix wrong lyrics or misaligned timestamps; the panel reloads on save
- =R= discard the cache and fetch again online
- =q= close the panel
- =?= list these keys

The keys are also shown at the top of the panel, and =L= is listed in simple-mpc's main screen.

Songs downloaded by org-songs carry no tags, so the artist falls back to the parent directory and the title to the file name.
Because the folder may not be an artist (e.g. =未归档=), each source is searched by title and artist first, then by title alone.
A hit must match the title; hits by the same artist, with the exact title, and not live are preferred.

#+begin_src emacs-lisp
  ;;; -*- lexical-binding: t -*-
  (require 'json)
  (require 'url)
  (require 'seq)

  (defvar my/lyrics-buffer "*simple-mpc-lyrics*")
  (defvar my/lyrics-music-dir (expand-file-name "~/code/org-songs/downloads/")
    "MPD music_directory; mpc reports song paths relative to it.")
  (defvar my/lyrics--current-song nil
    "(FILE ARTIST TITLE) of the song shown in the panel.")
  (defvar my/lyrics--idle-proc nil)

  (defvar my/lyrics-mode-map (make-sparse-keymap))
  ;; Bind outside the defvar so re-loading this file updates the keys of a running Emacs.
  (define-key my/lyrics-mode-map "g" #'my/lyrics-refresh)
  (define-key my/lyrics-mode-map "e" #'my/lyrics-edit)
  (define-key my/lyrics-mode-map "R" #'my/lyrics-refetch)
  ;; Close through the toggle so the mpc watcher stops too, unlike `quit-window'.
  (define-key my/lyrics-mode-map "q" #'my/lyrics-toggle)

  (define-derived-mode my/lyrics-mode special-mode "Lyrics"
    "Major mode for the simple-mpc lyrics panel.

  \\{my/lyrics-mode-map}")

  (defun my/lyrics--playing-song ()
    "Return (FILE ARTIST TITLE) of the playing song, or nil if nothing plays."
    (let ((out (string-trim-right
                (shell-command-to-string
                 "mpc -f '%file%\\t%artist%\\t%title%' current"))))
      (unless (string-empty-p out)
        (pcase-let ((`(,file ,artist ,title) (split-string out "\t")))
          (list file
                (if (string-empty-p (or artist ""))
                    (file-name-nondirectory (directory-file-name (or (file-name-directory file) "")))
                  artist)
                (if (string-empty-p (or title ""))
                    (file-name-base file)
                  title))))))

  (defun my/lyrics--song-base (file)
    "Absolute path of FILE without its extension, used to name lyrics files."
    (file-name-sans-extension (expand-file-name file my/lyrics-music-dir)))

  (defun my/lyrics--local-file (file)
    (let ((base (my/lyrics--song-base file)))
      (seq-find #'file-readable-p (list (concat base ".lrc") (concat base ".txt")))))

  (defun my/lyrics--strip-timestamps (text)
    "Remove LRC timestamps and ID tag lines such as [ti:...] from TEXT."
    (replace-regexp-in-string
     "^\\(\\[[0-9:.]+\\]\\)+" ""
     (replace-regexp-in-string "^\\[[a-z]+:[^]\n]*\\]\n" "" text)))

  (defun my/lyrics--read-local (path)
    (my/lyrics--strip-timestamps
     (with-temp-buffer
       (insert-file-contents path)
       (buffer-string))))

  (defun my/lyrics--render (header lyrics)
    (with-current-buffer (get-buffer-create my/lyrics-buffer)
      (unless (derived-mode-p 'my/lyrics-mode) (my/lyrics-mode))
      (let ((inhibit-read-only t))
        (erase-buffer)
        (insert (propertize header 'face 'bold) "\n"
                (propertize "g 刷新  e 编辑  R 重新获取  q 关闭  ? 帮助" 'face 'shadow) "\n\n"
                (or lyrics "(未找到歌词，按 e 手动添加)"))
        (goto-char (point-min)))))

  (defun my/lyrics--get-json (url callback &optional body)
    "Fetch URL and call CALLBACK with its parsed JSON, or nil on any failure.
  With BODY, a JSON-serializable object, send it as a POST request."
    (let ((url-request-method (if body "POST" "GET"))
          (url-request-data (when body (encode-coding-string (json-serialize body) 'utf-8)))
          ;; Set the agent here, not as an extra header: url.el would then send two User-Agent headers, which QQ Music rejects.
          (url-user-agent "Mozilla/5.0")
          (url-request-extra-headers `(;; QQ Music rejects requests without its own Referer.
                                       ("Referer" . "https://y.qq.com")
                                       ,@(when body '(("Content-Type" . "application/json"))))))
      (url-retrieve
       url
       (lambda (status)
         (let ((json (ignore-errors
                       (unless (plist-get status :error)
                         (goto-char url-http-end-of-headers)
                         (json-parse-buffer :object-type 'alist :array-type 'list
                                            :null-object nil :false-object nil)))))
           (kill-buffer)
           (funcall callback json)))
       ;; Inhibit cookies: once NetEase's NMTID cookie is sent back, its search returns unrelated songs.
       nil t t)))

  (defun my/lyrics--first (fetchers callback)
    "Run async FETCHERS in order until one yields non-nil, then call CALLBACK with it.
  Each fetcher takes a continuation; CALLBACK gets nil when all fail."
    (if (null fetchers)
        (funcall callback nil)
      (funcall (car fetchers)
               (lambda (result)
                 (if result
                     (funcall callback result)
                   (my/lyrics--first (cdr fetchers) callback))))))

  (defun my/lyrics--normalize (name)
    "Lowercase NAME without parenthesized parts and punctuation, for fuzzy title matching."
    (downcase (replace-regexp-in-string
               "[^[:alnum:]]+" ""
               (replace-regexp-in-string "[(（][^)）]*[)）]" "" name))))

  (defun my/lyrics--score (name singers artist title)
    "Rank a search hit called NAME by SINGERS for ARTIST and TITLE; nil when NAME does not match TITLE."
    (let ((norm (my/lyrics--normalize name))
          (want (my/lyrics--normalize title)))
      (when (and (> (length norm) 0) (> (length want) 0)
                 (or (string-search want norm) (string-search norm want)))
        (+ (if (member artist singers) 4 0)
           (if (equal norm want) 2 0)
           (if (string-match-p "live" (downcase name)) 0 1)))))

  (defun my/lyrics--try-best (hits artist title fetch-lyric callback)
    "Try FETCH-LYRIC on matching HITS best first; call CALLBACK with the first lyrics found.
  Each hit is (NAME SINGERS ID); FETCH-LYRIC takes an ID and a continuation."
    (let* ((scored (delq nil (mapcar (lambda (hit)
                                       (let ((score (my/lyrics--score (nth 0 hit) (nth 1 hit) artist title)))
                                         (when score (cons score (nth 2 hit)))))
                                     hits)))
           (ids (mapcar #'cdr (seq-sort-by #'car #'> scored))))
      (my/lyrics--first (mapcar (lambda (id) (lambda (k) (funcall fetch-lyric id k))) ids)
                        callback)))

  (defun my/lyrics--as-result (lrc)
    "Wrap LRC text as (SYNCED . PLAIN), or nil when it is blank."
    (when (and lrc (not (string-blank-p lrc)))
      (if (string-match-p "^\\[[0-9]" lrc) (cons lrc nil) (cons nil lrc))))

  (defun my/lyrics--netease (query artist title)
    "Return a fetcher searching NetEase Cloud Music for QUERY."
    (lambda (callback)
      (my/lyrics--get-json
       (concat "https://music.163.com/api/search/get?"
               (url-build-query-string `((s ,query) (type 1) (limit 10))))
       (lambda (json)
         (my/lyrics--try-best
          (mapcar (lambda (song)
                    (list (alist-get 'name song)
                          (mapcar (lambda (a) (alist-get 'name a)) (alist-get 'artists song))
                          (alist-get 'id song)))
                  (alist-get 'songs (alist-get 'result json)))
          artist title
          (lambda (id k)
            (my/lyrics--get-json
             (format "https://music.163.com/api/song/lyric?id=%s&lv=1" id)
             (lambda (json) (funcall k (my/lyrics--as-result (alist-get 'lyric (alist-get 'lrc json)))))))
          callback)))))

  (defun my/lyrics--qq (query artist title)
    "Return a fetcher searching QQ Music for QUERY."
    (lambda (callback)
      (my/lyrics--get-json
       "https://u.y.qq.com/cgi-bin/musicu.fcg"
       (lambda (json)
         (my/lyrics--try-best
          (mapcar (lambda (song)
                    (list (alist-get 'name song)
                          (mapcar (lambda (a) (alist-get 'name a)) (alist-get 'singer song))
                          (alist-get 'mid song)))
                  (let-alist json .req.data.body.song.list))
          artist title
          (lambda (mid k)
            (my/lyrics--get-json
             (concat "https://c.y.qq.com/lyric/fcgi-bin/fcg_query_lyric_new.fcg?"
                     (url-build-query-string `((songmid ,mid) (format "json") (nobase64 1))))
             (lambda (json) (funcall k (my/lyrics--as-result (alist-get 'lyric json))))))
          callback))
       `((req . ((module . "music.search.SearchCgiService")
                 (method . "DoSearchForQQMusicDesktop")
                 (param . ((query . ,query) (num_per_page . 10) (page_num . 1) (search_type . 0)))))))))

  (defun my/lyrics--fetch-online (artist title callback)
    "Call CALLBACK with (SYNCED . PLAIN) lyrics for ARTIST and TITLE, or nil if no source has them."
    (my/lyrics--first
     (list (my/lyrics--netease (format "%s %s" title artist) artist title)
           (my/lyrics--netease title artist title)
           (my/lyrics--qq (format "%s %s" title artist) artist title)
           (my/lyrics--qq title artist title))
     callback))

  (defun my/lyrics--save-cache (file result)
    "Write RESULT, a (SYNCED . PLAIN) pair, next to song FILE and return the written path."
    (let* ((synced (car result))
           (path (concat (my/lyrics--song-base file) (if synced ".lrc" ".txt"))))
      (with-temp-file path
        (insert (or synced (cdr result))))
      path))

  (defun my/lyrics--show (song &optional skip-local)
    "Display lyrics of SONG, a (FILE ARTIST TITLE) list; with SKIP-LOCAL ignore the cache."
    (pcase-let* ((`(,file ,artist ,title) song)
                 (header (format "%s - %s" artist title))
                 (local (unless skip-local (my/lyrics--local-file file))))
      (setq my/lyrics--current-song song)
      (if local
          (my/lyrics--render header (my/lyrics--read-local local))
        (my/lyrics--render header "加载中...")
        (my/lyrics--fetch-online
         artist title
         (lambda (result)
           ;; Drop stale responses if the song changed meanwhile.
           (when (equal song my/lyrics--current-song)
             (if result
                 (my/lyrics--render header (my/lyrics--read-local (my/lyrics--save-cache file result)))
               (my/lyrics--render header nil))))))))

  (defun my/lyrics-refresh (&optional force)
    "Show lyrics of the playing song; skip when the song is unchanged unless FORCE."
    (interactive (list t))
    (let ((song (my/lyrics--playing-song)))
      (cond ((null song)
             (setq my/lyrics--current-song nil)
             (my/lyrics--render "未在播放" ""))
            ((or force (not (equal song my/lyrics--current-song)))
             (my/lyrics--show song)))))

  (defun my/lyrics-edit ()
    "Edit the lyrics file of the shown song, creating a .lrc if none exists."
    (interactive)
    (let ((file (car my/lyrics--current-song)))
      (unless file (user-error "No song is playing"))
      (find-file-other-window
       (or (my/lyrics--local-file file) (concat (my/lyrics--song-base file) ".lrc")))
      (add-hook 'after-save-hook (lambda () (my/lyrics-refresh t)) nil t)))

  (defun my/lyrics-refetch ()
    "Delete the cached lyrics of the shown song and fetch them again online."
    (interactive)
    (let* ((song my/lyrics--current-song)
           (local (and song (my/lyrics--local-file (car song)))))
      (unless song (user-error "No song is playing"))
      (when (and local (not (yes-or-no-p (format "Overwrite %s? " (file-name-nondirectory local)))))
        (user-error "Cancelled"))
      (when local (delete-file local))
      (my/lyrics--show song t)))

  (defun my/lyrics--start-watch ()
    "Refresh lyrics whenever MPD's player state changes."
    (unless (process-live-p my/lyrics--idle-proc)
      (setq my/lyrics--idle-proc
            (make-process :name "mpc-idle"
                          :command '("mpc" "idleloop" "player")
                          :noquery t
                          :filter (lambda (_proc _out) (my/lyrics-refresh))))))

  (defun my/lyrics--stop-watch ()
    (when (process-live-p my/lyrics--idle-proc)
      (delete-process my/lyrics--idle-proc)))

  (defun my/lyrics-toggle ()
    "Toggle the lyrics panel on the right half of the frame."
    (interactive)
    (let ((win (get-buffer-window my/lyrics-buffer)))
      (if win
          (progn (delete-window win) (my/lyrics--stop-watch))
        (my/lyrics-refresh t)
        (display-buffer (get-buffer-create my/lyrics-buffer)
                        '(display-buffer-in-side-window (side . right) (window-width . 0.5)))
        (my/lyrics--start-watch))))

  (with-eval-after-load 'simple-mpc-mode
    (define-key simple-mpc-mode-map (kbd "L") #'my/lyrics-toggle))

  (defun my/lyrics--add-to-main-screen (&rest _)
    "List the lyrics key in simple-mpc's hard-coded main screen menu."
    (when-let ((buf (get-buffer simple-mpc-main-buffer-name)))
      (with-current-buffer buf
        (let ((inhibit-read-only t))
          (goto-char (point-max))
          (insert (propertize "\n\n   * lyrics\n" 'face 'simple-mpc-main-headers)
                  "      * toggle [L]yrics panel")
          (goto-char (point-min))))))

  (advice-add 'simple-mpc :after #'my/lyrics--add-to-main-screen)
#+end_src
